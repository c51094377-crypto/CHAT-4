        let isCamOff = false;

            document.getElementById('settings-name-input').value = myName;

        function updateMyAvatarDisplay() {
            document.getElementById('display-name').innerText = myName;
            const container = document.getElementById('user-avatar-container');
            if (myAvatar) {
                container.innerHTML = `<img src="${myAvatar}" class="w-full h-full object-cover">`;
            } else {
                container.innerHTML = `<span id="user-avatar-text">${myName.charAt(0)}</span>`;
            }
        }

        function openSettings() {
            document.getElementById('settings-name-input').value = myName;
            document.getElementById('settings-modal').classList.remove('hidden');
        }

        function closeSettings() {
            document.getElementById('settings-modal').classList.add('hidden');
        }

        function saveSettings() {
            const nameInput = document.getElementById('settings-name-input').value.trim();
            if (nameInput) {
                myName = nameInput;
                localStorage.setItem('nexus_my_name', myName);
            }

            const avatarFile = document.getElementById('settings-avatar-input').files[0];
            if (avatarFile) {
                const reader = new FileReader();
                reader.onload = function(e) {
                    myAvatar = e.target.result;
                    localStorage.setItem('nexus_my_avatar', myAvatar);
                    updateMyAvatarDisplay();
                    closeSettings();
                };
                reader.readAsDataURL(avatarFile);
            } else {
                updateMyAvatarDisplay();
                closeSettings();
            }
        }

        function toggleSidebar() {
            const sidebar = document.getElementById('sidebar');
            if (sidebar.style.display === 'flex' && window.innerWidth < 768) {
                sidebar.style.display = 'none';
            } else {
                sidebar.style.display = 'flex';
                sidebar.classList.remove('hidden');
            }
        }

        window.addEventListener('resize', () => {
            const sidebar = document.getElementById('sidebar');
            if (window.innerWidth >= 768) {
                sidebar.style.display = 'flex';
                sidebar.classList.remove('hidden');
            } else {
                sidebar.style.display = 'none';
                sidebar.classList.add('hidden');
            }
        });

        function saveRoomToHistory(modeName, roomId) {
            let history = JSON.parse(localStorage.getItem('nexus_room_history') || '[]');
            history = history.filter(item => item.roomId !== roomId);
            history.unshift({ modeName, roomId, time: new Date().toLocaleDateString() });
            if (history.length > 10) history.pop();
            localStorage.setItem('nexus_room_history', JSON.stringify(history));
            renderHistoryList();
        }

        function renderHistoryList() {
            const container = document.getElementById('history-list');
            const history = JSON.parse(localStorage.getItem('nexus_room_history') || '[]');
            if (history.length === 0) {
                container.innerHTML = '<p class="text-xs text-slate-500 text-center py-2">記録されたルームはありません</p>';
                return;
            }
            container.innerHTML = '';
            history.forEach(item => {
                const div = document.createElement('div');
                div.className = 'flex items-center justify-between bg-slate-800 hover:bg-slate-750 p-2.5 rounded-xl border border-slate-700 text-xs transition';
                div.innerHTML = `
                    <div>
                        <span class="font-bold text-white">${item.modeName}</span>
                        <span class="text-indigo-400 ml-1.5 tracking-wider">[${item.roomId}]</span>
                    </div>
                    <button onclick="reconnectRoom('${item.modeName}', '${item.roomId}')" class="bg-indigo-600 hover:bg-indigo-500 text-white px-2.5 py-1 rounded-lg font-semibold transition">再接続</button>
                `;
                container.appendChild(div);
            });
        }

        function clearHistory() {
            localStorage.removeItem('nexus_room_history');
            renderHistoryList();
        }

        function reconnectRoom(modeName, roomId) {
            const modeType = (modeName === 'ダイレクト') ? 'private' : 'group';
            if (modeType === 'private') {
                switchMode('private');
                document.getElementById('private-room-input').value = roomId;
            } else {
                switchMode('group');
                document.getElementById('group-room-input').value = roomId;
            }
            initRoom(modeName, roomId, modeType);
        }

        function initRoom(modeName, roomId, modeType) {
            if (peer) {
                try {
                    peer.destroy();
                } catch (e) {
                    console.error(e);
                }
                peer = null;
            }
            stopAllCalls();
            connections = {};
            incomingBuffers = {};
            activePeerIds.clear();
            chatZipFiles = [];
            currentRoomId = roomId;
            currentModeName = modeName;
            currentMode = modeType;

            saveRoomToHistory(modeName, roomId);

            document.getElementById('chat-room-title').innerText = `${modeName} (コード: ${currentRoomId})`;
            document.getElementById('chat-room-subtitle').innerText = '接続準備中...';
            document.getElementById('connection-status').innerHTML = '<span class="w-2.5 h-2.5 rounded-full bg-blue-500 animate-pulse"></span><span>接続確立中</span>';

            clearMessages();
            appendSystemMessage(`ルーム [コード: ${currentRoomId}] (${modeName}) に接続を試みています...`);

            const prefix = (modeType === 'private') ? 'nexus_private' : 'nexus_group';
            const hostPeerId = `${prefix}_${roomId}_host`;
            const uniqueId = Math.random().toString(36).substring(2, 6);
            const myId = `${prefix}_${roomId}_user_${uniqueId}`;

            peer = new Peer(hostPeerId, peerConfig);

            peer.on('open', (id) => {
                document.getElementById('chat-room-subtitle').innerText = (modeType === 'private') ? 'ダイレクト待受中' : 'スクール空間ホスト（最大40人対応）';
                appendSystemMessage((modeType === 'private') ? 'ダイレクト回線を開設しました。相手の接続を待っています...' : 'スクール空間（最大40人対応）を開設しました。');
                document.getElementById('connection-status').innerHTML = '<span class="w-2.5 h-2.5 rounded-full bg-amber-500"></span><span>待受中</span>';
                
                activePeerIds.add(id);

                peer.on('connection', (conn) => {
                    setupConnection(conn, modeType);
                    broadcastPeerList();
                });

                peer.on('call', (call) => {
                    handleIncomingCall(call);
                });
            });

            peer.on('error', (err) => {
                if (err.type === 'unavailable-id') {
                    peer.destroy();
                    
                    peer = new Peer(myId, peerConfig);
                    peer.on('open', (openedId) => {
                        appendSystemMessage('スクール空間を発見しました。最大40人のネットワークに参加中...');
                        activePeerIds.add(openedId);
                        
                        const conn = peer.connect(hostPeerId);
                        setupConnection(conn, modeType);

                        setTimeout(() => {
                            if (conn && conn.open) {
                                conn.send({ type: 'request_peer_list', from: openedId });
                            }
                        }, 800);

                        setTimeout(() => {
                            if (localStream) {
                                startCall(hostPeerId);
                            }
                        }, 1000);
                    });

                    peer.on('connection', (conn) => {
                        setupConnection(conn, modeType);
                    });

                    peer.on('call', (call) => {
                        handleIncomingCall(call);
                    });

                    peer.on('error', (subErr) => {
                        console.error(subErr);
                        document.getElementById('connection-status').innerHTML = '<span class="w-2.5 h-2.5 rounded-full bg-red-500"></span><span>接続エラー (network)</span>';
                        appendSystemMessage(`ネットワークエラー: ${subErr.type}`);
                    });
                } else {
                    console.error(err);
                    document.getElementById('connection-status').innerHTML = '<span class="w-2.5 h-2.5 rounded-full bg-red-500"></span><span>ネットワークエラー</span>';
                    appendSystemMessage(`ネットワークエラー: ${err.type}`);
                }
            });
        }

        function setupConnection(conn, modeType) {
            connections[conn.peer] = conn;
            activePeerIds.add(conn.peer);

            conn.on('open', () => {
                document.getElementById('connection-status').innerHTML = '<span class="w-2.5 h-2.5 rounded-full bg-emerald-500"></span><span>接続済み</span>';
                appendSystemMessage('新しいメンバーがスクール空間に参加しました');
                updatePeerCount();

                if ((isVideoActive || isAudioOnlyActive) && localStream) {
                    startCall(conn.peer);
                }
                
                conn.send({ type: 'read_receipt', reader: myName });
            });

            conn.on('close', () => {
                delete connections[conn.peer];
                activePeerIds.delete(conn.peer);
                document.getElementById('connection-status').innerHTML = '<span class="w-2.5 h-2.5 rounded-full bg-amber-500"></span><span>メンバー退出</span>';
                appendSystemMessage('メンバーが退出しました');
                updatePeerCount();
            });

            conn.on('data', (packet) => {
                handleDataPacket(conn.peer, packet, modeType);
            });
        }

        function broadcastPeerList() {
            const peerList = Array.from(activePeerIds);
            if (peerList.length > 40) {
                peerList.length = 40;
            }
            Object.keys(connections).forEach(peerId => {
                const conn = connections[peerId];
                if (conn && conn.open) {
                    conn.send({ type: 'peer_list', peers: peerList });
                }
            });
        }

        function connectToPeers(peerList) {
            peerList.forEach(id => {
                if (id !== peer.id && !connections[id] && activePeerIds.size < 40) {
                    const conn = peer.connect(id);
                    setupConnection(conn, currentMode);
                }
            });
        }

        function handleDataPacket(peerId, packet, modeType) {
            if (packet.type === 'peer_list') {
                connectToPeers(packet.peers);
                return;
            }
            if (packet.type === 'request_peer_list') {
                broadcastPeerList();
                return;
            }
            if (packet.type === 'read_receipt') {
                markAllMyMessagesAsRead();
                if (modeType === 'group') {
                    forwardToGroupOthers(peerId, packet);
                }
                return;
            }

            if (!incomingBuffers[peerId]) {
                incomingBuffers[peerId] = {};
            }

            if (packet.type === 'chat') {
                appendMessage(packet.senderName, packet.senderAvatar, packet.text, false, packet.timestamp, null, packet.isSticker, packet.isVideo, packet.mediaData, packet.isZip, packet.zipName);
                sendPayloadReliably({ type: 'read_receipt', reader: myName });

                if (modeType === 'group') {
                    forwardToGroupOthers(peerId, packet);
                }
            } else if (packet.type === 'chunk_start') {
                incomingBuffers[peerId][packet.fileId] = {
                    senderName: packet.senderName,
                    senderAvatar: packet.senderAvatar,
                    timestamp: packet.timestamp,
                    totalChunks: packet.totalChunks,
                    receivedChunks: 0,
                    chunks: new Array(packet.totalChunks),
                    isVideo: packet.isVideo,
                    isZip: packet.isZip,
                    zipName: packet.zipName
                };
                sendPayloadReliably({ type: 'read_receipt', reader: myName });
                if (modeType === 'group') {
                    forwardToGroupOthers(peerId, packet);
                }
            } else if (packet.type === 'chunk_data') {
                const fileBuf = incomingBuffers[peerId][packet.fileId];
                if (fileBuf && !fileBuf.chunks[packet.chunkIndex]) {
                    fileBuf.chunks[packet.chunkIndex] = packet.data;
                    fileBuf.receivedChunks++;

                    if (fileBuf.receivedChunks === fileBuf.totalChunks) {
                        const fullMediaBase64 = fileBuf.chunks.join('');
                        appendMessage(fileBuf.senderName, fileBuf.senderAvatar, '', false, fileBuf.timestamp, fullMediaBase64, false, fileBuf.isVideo, null, fileBuf.isZip, fileBuf.zipName);
                        delete incomingBuffers[peerId][packet.fileId];
                    }
                }
                if (modeType === 'group') {
                    forwardToGroupOthers(peerId, packet);
                }
            }
        }

        let relayedPackets = new Set();
        function forwardToGroupOthers(excludePeerId, packet) {
            const packetKey = (packet.senderName || '') + (packet.text || packet.fileId || packet.type) + (packet.timestamp || '');
            if (packet.type === 'read_receipt') {
                // pass
            } else {
                if (relayedPackets.has(packetKey)) return;
                relayedPackets.add(packetKey);
                if (relayedPackets.size > 200) relayedPackets.clear();
            }

            Object.keys(connections).forEach(targetId => {
                if (targetId !== excludePeerId) {
                    const conn = connections[targetId];
                    if (conn && conn.open) {
                        conn.send(packet);
                    }
                }
            });
        }

        function updatePeerCount() {
            const count = Object.keys(connections).filter(id => connections[id].open).length + 1;
            document.getElementById('peer-count').innerText = `参加者: ${Math.min(count, 40)}人`;
            document.getElementById('chat-room-subtitle').innerText = `接続中 (${Math.min(count, 40)}人と同期中)`;
        }

        function sendPayloadReliably(payloadObj) {
            Object.keys(connections).forEach(peerId => {
                const conn = connections[peerId];
                if (conn && conn.open) {
                    conn.send(payloadObj);
                }
            });
        }

        function sendMessage() {
            const input = document.getElementById('message-input');
            const text = input.value.trim();
            if (!text) return;

            const timestamp = new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
            appendMessage(myName, myAvatar, text, true, timestamp, null, false, false, null, false, null);
            
            const packetKey = myName + text + timestamp;
            relayedPackets.add(packetKey);

            sendPayloadReliably({
                type: 'chat',
                senderName: myName,
                senderAvatar: myAvatar,
                text: text,
                timestamp: timestamp,
                isSticker: false,
                isVideo: false,
                isZip: false
            });

            input.value = '';
        }

        async function sendMediaFiles(event) {
            const files = event.target.files;
            if (!files || files.length === 0) return;

            // 350MB制限チェック
            let totalSize = 0;
            for (let i = 0; i < files.length; i++) {
                totalSize += files[i].size;
            }
            if (totalSize > 350 * 1024 * 1024) {
                alert('ファイルサイズの合計が350MBを超えています。350MB以下のファイルを選択してください。');
                event.target.value = '';
                return;
            }

            try {
                const zip = new JSZip();
                const zipName = 'media_bundle_350mb_' + Math.floor(Date.now() / 1000) + '.zip';

                for (let i = 0; i < files.length; i++) {
                    const file = files[i];
                    zip.file(file.name, file);
                }

                // 350MBの大容量に対応するためストリーム・圧縮オプションを最適化
                const zipBlob = await zip.generateAsync({ type: 'base64', compression: 'DEFLATE', compressionOptions: { level: 3 } });
                const zipBase64 = 'data:application/zip;base64,' + zipBlob;
                const timestamp = new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });

                appendMessage(myName, myAvatar, '', true, timestamp, zipBase64, false, false, null, true, zipName);
                chatZipFiles.push({ name: zipName, data: zipBase64 });

                const fileId = 'zip_' + Math.random().toString(36).substring(2, 9);
                relayedPackets.add(myName + fileId + timestamp);

                const CHUNK_SIZE = 8 * 1024; // 350MBの安定転送用ミニチャンク
                const totalChunks = Math.ceil(zipBase64.length / CHUNK_SIZE);

                sendPayloadReliably({
                    type: 'chunk_start',
                    fileId: fileId,
                    senderName: myName,
                    senderAvatar: myAvatar,
                    timestamp: timestamp,
                    totalChunks: totalChunks,
                    isVideo: false,
                    isZip: true,
                    zipName: zipName
                });

                let currentChunk = 0;
                function sendNextChunk() {
                    if (currentChunk < totalChunks) {
                        const start = currentChunk * CHUNK_SIZE;
                        const chunkData = zipBase64.substring(start, start + CHUNK_SIZE);

                        sendPayloadReliably({
                            type: 'chunk_data',
                            fileId: fileId,
                            chunkIndex: currentChunk,
                            data: chunkData
                        });

                        currentChunk++;
                        setTimeout(sendNextChunk, 40);
                    }
                }
                sendNextChunk();
            } catch (err) {
                console.error(err);
                alert('350MBまでのファイルZIP圧縮・送信に失敗しました。');
            }
            event.target.value = '';
        }

        async function downloadAllZips() {
            if (chatZipFiles.length === 0) {
                alert('ダウンロードできるZIPファイルはありません。');
                return;
            }
            try {
                const masterZip = new JSZip();
                chatZipFiles.forEach((zf, idx) => {
                    const base64Content = zf.data.split(',')[1];
                    masterZip.file(zf.name || `bundle_${idx}.zip`, base64Content, { base64: true });
                });

                const content = await masterZip.generateAsync({ type: 'blob' });
                const url = URL.createObjectURL(content);
                const a = document.createElement('a');
                a.href = url;
                a.download = 'all_chat_bundles_350mb.zip';
                document.body.appendChild(a);
                a.click();
                document.body.removeChild(a);
                URL.revokeObjectURL(url);
            } catch (err) {
                console.error(err);
                alert('一括ZIPダウンロードに失敗しました。');
            }
        }

        function appendMessage(sender, senderAvatar, text, isMe, timestamp, media = null, isSticker = false, isVideo = false, mediaData = null, isZip = false, zipName = null) {
            const container = document.getElementById('messages-container');
            const messageRow = document.createElement('div');
            messageRow.className = `flex items-end space-x-2 mb-3 ${isMe ? 'justify-end' : 'justify-start'} chat-message-row`;

            let avatarHtml = '';
            if (senderAvatar) {
                avatarHtml = `<div class="w-8 h-8 rounded-full overflow-hidden flex-shrink-0 shadow border border-indigo-500"><img src="${senderAvatar}" class="w-full h-full object-cover"></div>`;
            } else {
                const initial = sender ? sender.charAt(0) : 'U';
                avatarHtml = `<div class="w-8 h-8 bg-indigo-600 text-white rounded-full flex items-center justify-center font-bold text-xs flex-shrink-0 shadow">${initial}</div>`;
            }

            let contentHtml = '';
            if (media) {
                if (isZip) {
                    if (media && !chatZipFiles.some(z => z.data === media)) {
                        chatZipFiles.push({ name: zipName || 'bundle.zip', data: media });
                    }
                    contentHtml += `
                        <div class="bg-slate-900 border border-slate-700 p-3 rounded-xl flex items-center space-x-3 mb-1">
                            <i class="fa-solid fa-file-zipper text-amber-400 text-2xl"></i>
                            <div>
                                <p class="text-xs font-bold text-white">${escapeHtml(zipName || 'bundle.zip')}</p>
                                <a href="${media}" download="${zipName || 'bundle.zip'}" class="text-[10px] text-indigo-400 hover:underline">個別にダウンロード</a>
                            </div>
                        </div>
                    `;
                } else if (isVideo) {
                    contentHtml += `<video src="${media}" controls class="max-w-xs rounded-xl mb-1 shadow border border-slate-700"></video>`;
                } else {
                    contentHtml += `<img src="${media}" class="max-w-xs rounded-xl mb-1 shadow border border-slate-700">`;
                }
            }
            if (text) {
                if (isSticker) {
                    contentHtml += `<div class="text-sm font-bold py-1.5 px-3 bg-indigo-950 text-indigo-200 border border-indigo-700/50 rounded-xl shadow-inner">${escapeHtml(text)}</div>`;
                } else {
                    contentHtml += `<p class="text-sm text-slate-100">${escapeHtml(text)}</p>`;
                }
            }

            if (isMe) {
                messageRow.innerHTML = `
                    <div class="flex items-end space-x-1 max-w-xs md:max-w-md">
                        <div class="flex flex-col items-end self-end mb-0.5">
                            <span class="text-[10px] text-emerald-400 font-bold read-status hidden">既読</span>
                            <span class="text-[10px] text-slate-400">${timestamp}</span>
                        </div>
                        <div class="bg-indigo-600 text-white rounded-2xl px-4 py-2.5 shadow-md rounded-br-none">
                            ${contentHtml}
                        </div>
                    </div>
                    ${avatarHtml}
                `;
            } else {
                messageRow.innerHTML = `
                    ${avatarHtml}
                    <div class="flex flex-col">
                        <span class="text-xs text-slate-400 ml-1 mb-1">${escapeHtml(sender)}</span>
                        <div class="flex items-end space-x-1 max-w-xs md:max-w-md">
                            <div class="bg-slate-800 border border-slate-700 text-slate-100 rounded-2xl px-4 py-2.5 shadow-md rounded-bl-none">
                                ${contentHtml}
                            </div>
                            <span class="text-[10px] text-slate-400 self-end mb-1">${timestamp}</span>
                        </div>
                    </div>
                `;
            }

            container.appendChild(messageRow);
            container.scrollTop = container.scrollHeight;
        }
